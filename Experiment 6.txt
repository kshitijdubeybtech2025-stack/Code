script.js
// EXPERIMENT 6: JavaScript validation and secure form handling

// 1. Sanitize input by escaping HTML-special characters.
function sanitize(str) {
    const div = document.createElement("div");
    div.textContent = String(str);
    return div.innerHTML;
}

// 2. Name validation: letters and spaces, 2–40 characters.
function isValidName(value) {
    return /^[A-Za-z ]{2,40}$/.test(value.trim());
}

// 3. Email validation.
function isValidEmail(value) {
    return /^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(value.trim());
}

// 4. Age validation: whole number between 1 and 100.
function isValidAge(value) {
    if (value.trim() === "") {
        return false;
    }

    const age = Number(value);
    return Number.isInteger(age) && age >= 1 && age <= 100;
}

// 5. Password: minimum 8 characters, one number,
//    and at least one non-alphanumeric character.
function isValidPassword(value) {
    return /^(?=.*\d)(?=.*[^A-Za-z0-9]).{8,}$/.test(value);
}

// 6. Live email validation using the input event.
const emailInput = document.getElementById("email");
const emailStatus = document.getElementById("emailStatus");

emailInput.addEventListener("input", function () {
    const email = emailInput.value.trim();

    emailStatus.classList.remove("valid", "invalid");

    if (email === "") {
        emailStatus.textContent = "Enter your email address.";
    } else if (isValidEmail(email)) {
        emailStatus.textContent = "Valid email";
        emailStatus.classList.add("valid");
    } else {
        emailStatus.textContent = "Invalid email";
        emailStatus.classList.add("invalid");
    }
});

// 7. Register button: validate each field and stop at the first error.
document.getElementById("submitBtn").addEventListener("click", function () {
    const name = document.getElementById("name").value;
    const email = document.getElementById("email").value;
    const age = document.getElementById("age").value;
    const password = document.getElementById("password").value;
    const confirmPassword =
        document.getElementById("confirmPassword").value;
    const bio = document.getElementById("bio").value;

    if (!isValidName(name)) {
        alert("Please enter a valid name using 2–40 letters and spaces.");
        return;
    }

    if (!isValidEmail(email)) {
        alert("Please enter a valid email address.");
        return;
    }

    if (!isValidAge(age)) {
        alert("Age must be a whole number between 1 and 100.");
        return;
    }

    if (!isValidPassword(password)) {
        alert("Password must have at least 8 characters, one number, and one special character.");
        return;
    }

    if (confirmPassword !== password) {
        alert("Password and Confirm Password do not match.");
        return;
    }

    // Safe output: escape user input before inserting it as HTML.
    // Never insert untrusted input directly through innerHTML.
    const output = document.getElementById("output");

    output.innerHTML =
        "Welcome, " + sanitize(name.trim()) +
        "! Your bio: " + sanitize(bio);

    alert("Registration form validated successfully.");
});

// 8. Change Nickname using prompt().
document.getElementById("nickBtn").addEventListener("click", function () {
    const nickname = prompt("Enter a nickname:");

    if (nickname !== null && nickname.trim() !== "") {
        // textContent safely displays user input as text.
        document.getElementById("nickOutput").textContent =
            "Nickname set to: " + nickname.trim();
    } else if (nickname !== null) {
        alert("Nickname cannot be empty.");
    }
});

// 9. Clear Form using confirm().
document.getElementById("clearBtn").addEventListener("click", function () {
    if (confirm("Are you sure you want to clear the form?")) {
        document.getElementById("regForm").reset();

        document.getElementById("output").textContent = "";
        document.getElementById("nickOutput").textContent = "";
        document.getElementById("feedbackOutput").textContent = "";

        emailStatus.textContent = "Enter your email address.";
        emailStatus.classList.remove("valid", "invalid");
    }
});

// 10. Quick Feedback using prompt().
// Repeats the prompt if the rating is invalid.
document.getElementById("feedbackBtn").addEventListener("click", function () {
    while (true) {
        const answer = prompt("Rate this form from 1 to 5:");

        // Stop if the user presses Cancel.
        if (answer === null) {
            return;
        }

        const ratingText = answer.trim();

        if (/^[1-5]$/.test(ratingText)) {
            document.getElementById("feedbackOutput").textContent =
                "Thank you for your feedback! Your rating: " +
                ratingText + "/5.";
            break;
        }

        alert("Invalid rating. Please enter a whole number from 1 to 5.");
    }
});
