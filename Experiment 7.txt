const http = require('http');
const server = http.createServer((req, res) => {
    res.setHeader('Content-Type', 'text/html');
    if (req.url === '/') {
        res.statusCode = 200;
        res.end('<h1>Home Page</h1><p>Welcome to my Node.js server</p>');
    } else if (req.url === '/about') {
        res.statusCode = 200;
        res.end('<h1>About Page</h1><p>This server demonstrates the HTTP module.</p>');
    } else if (req.url === '/contact') {
        res.statusCode = 200;
        res.end('<h1>Contact Page</h1><p>Email: student@example.com</p>');
    } else {
        res.statusCode = 404;
        res.end('<h1>404 - Page Not Found</h1>');
    }
});
const PORT = 3000;
server.listen(PORT, () => {
    console.log('Server running at http://localhost:' + PORT);
});

server.js
const http = require('http');
const greetings = require('./greetings');

const server = http.createServer((req, res) => {
    res.setHeader('Content-Type', 'text/html; charset=utf-8');

    if (req.method !== 'GET') {
        res.statusCode = 405;
        return res.end('<h1>405 - Method Not Allowed</h1>');
    }

    if (req.url === '/') {
        res.statusCode = 200;
        res.end(`
            <html>
            <body style="font-family:Arial;text-align:center;background:#e3f2fd;padding:40px">
                <h1>Home Page</h1>
                <p>Welcome to my Node.js server!</p>
                <a href="/about">About</a> |
                <a href="/contact">Contact</a> |
                <a href="/services">Services</a>
            </body>
            </html>
        `);
    } else if (req.url === '/about') {
        res.statusCode = 200;
        res.end(`
            <html><body style="font-family:Arial;background:#e8f5e9;padding:40px">
                <h1>About Us</h1>
                <p>This website demonstrates Node.js modules and HTTP routing.</p>
            </body></html>
        `);
    } else if (req.url === '/contact') {
        res.statusCode = 200;
        res.end(`
            <html><body style="font-family:Arial;background:#fff3e0;padding:40px">
                <h1>Contact Us</h1>
                <p>Email: student@example.com</p>
            </body></html>
        `);
    } else if (req.url === '/services') {
        res.statusCode = 200;
        res.end(`
            <html><body style="font-family:Arial;background:#f3e5f5;padding:40px">
                <h1>Our Services</h1>
                <ul><li>Web Development</li><li>Node.js Applications</li><li>Database Integration</li></ul>
            </body></html>
        `);
    } else if (req.url === '/greetings') {
        res.statusCode = 200;
        res.end(`
            <html><body style="font-family:Arial;background:#fffde7;padding:40px">
                <h1>${greetings.morningGreeting('Kshitij')}</h1>
                <p>Formatted name: ${greetings.formatName('Kshitij')}</p>
                <p>${greetings.eveningGreeting('Kshitij')}</p>
            </body></html>
        `);
    } else {
        res.statusCode = 404;
        res.end(`
            <html><body style="font-family:Arial;text-align:center;padding:40px">
                <h1>404 - Page Not Found</h1>
                <p>The requested page does not exist.</p>
            </body></html>
        `);
    }
});

const PORT = 3000;

server.listen(PORT, () => {
    console.log('Server running at http://localhost:' + PORT);
});

        res.statusCode = 200;
        res.end('<html><body style="font-family:Arial;background:#f3e5f5;padding:40px"><h1>Our Services</h1><ul><li>Web Development</li><li>Node.js Applications</li><li>Database Integration</li></ul></body></html>');

        `);

        res.statusCode = 200;
        res.end(`
            <html><body style="font-family:Arial;padding:40px">
                <h1>Current Server Date and Time</h1>
